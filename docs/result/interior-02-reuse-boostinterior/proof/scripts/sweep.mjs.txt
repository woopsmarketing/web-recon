// Scratch tooling (not a repo file): master's final browser smoke over every route x width.
//   node sweep.mjs <baseUrl> <out.json> [--shots <dir>]
// Per page x width: HTTP status, console errors, page errors, failed requests (prefetch aborts ignored),
// broken images (after scrolling the whole page so lazy images load), off-origin requests, horizontal
// overflow, section roots. Then: an internal-link crawl (every same-origin href found on any page must
// answer 200) and a forbidden-term scan of the rendered DOM text.
import { chromium } from "/Users/woops/projects/web-recon-track-b/node_modules/playwright/index.mjs";
import { mkdirSync, writeFileSync } from "node:fs";

const [base, outFile] = process.argv.slice(2);
const shotsDir = process.argv.includes("--shots") ? process.argv[process.argv.indexOf("--shots") + 1] : null;
if (shotsDir) mkdirSync(shotsDir, { recursive: true });
const ROUTES = process.env.ROUTES.split(",");
const WIDTHS = [320, 390, 768, 1024, 1280, 1440, 1920];
const TERMS = process.env.TERMS.split(",");
const ALLOW_OFF = (process.env.ALLOW_OFF ?? "").split(",").filter(Boolean);
const BLOCK = (process.env.BLOCK ?? "").split(",").filter(Boolean);
const origin = new URL(base).origin;

const browser = await chromium.launch();
const rows = [];
const hrefs = new Set();
for (const width of WIDTHS) {
  const mobile = width <= 640;
  const context = await browser.newContext({
    viewport: { width, height: mobile ? 844 : 900 },
    deviceScaleFactor: 1,
    isMobile: mobile,
    hasTouch: mobile,
    locale: "ko-KR",
    timezoneId: "Asia/Seoul",
  });
  for (const b of BLOCK) await context.route(b, (r) => r.abort("aborted"));
  for (const route of ROUTES) {
    const page = await context.newPage();
    const consoleErrors = [];
    const pageErrors = [];
    const failed = [];
    const offOrigin = new Set();
    page.on("console", (m) => m.type() === "error" && consoleErrors.push(m.text().slice(0, 200)));
    page.on("pageerror", (e) => pageErrors.push(String(e).slice(0, 200)));
    page.on("requestfailed", (r) => {
      const why = r.failure()?.errorText ?? "";
      if (why.includes("ERR_ABORTED")) return; // Next <Link> prefetches cancelled by navigation/teardown
      failed.push(`${r.url().slice(0, 120)} ${why}`);
    });
    page.on("request", (r) => {
      const u = r.url();
      if (!u.startsWith(origin) && !u.startsWith("data:") && !u.startsWith("blob:")) offOrigin.add(new URL(u).origin);
    });
    const res = await page.goto(base + route, { waitUntil: "networkidle" });
    const status = res?.status() ?? 0;
    // before any scrolling: an element parked outside the viewport (a waiting reveal) must not widen the layout viewport
    const atLoad = await page.evaluate(() => ({ inner: innerWidth, scroll: document.documentElement.scrollWidth }));
    const loadWide = atLoad.inner !== width || atLoad.scroll !== width ? atLoad : null;
    // walk the page so every lazy image is requested
    const docHeight = await page.evaluate(() => document.documentElement.scrollHeight);
    for (let y = 0; y < docHeight; y += 700) {
      await page.evaluate((y) => scrollTo(0, y), y);
      await page.waitForTimeout(60);
    }
    await page.waitForLoadState("networkidle");
    await page.evaluate(() => scrollTo(0, 0));
    const data = await page.evaluate((terms) => {
      const de = document.documentElement;
      const broken = [...document.images].filter((i) => i.complete && i.naturalWidth === 0 && i.getAttribute("src")).map((i) => i.getAttribute("src"));
      // not loaded although laid out inside the viewport column (lazy images in hidden/off-canvas copies are expected to stay unloaded)
      const pending = [...document.images].filter((i) => { if (i.complete) return false; const r = i.getBoundingClientRect(); return i.getClientRects().length > 0 && r.width > 0 && r.right > 0 && r.left < de.clientWidth; }).length;
      // elements that stick out of the viewport horizontally (ignoring ones inside clipped/scrollable ancestors)
      const wide = [];
      if (de.scrollWidth > de.clientWidth) {
        for (const e of document.querySelectorAll("body *")) {
          const r = e.getBoundingClientRect();
          if (r.right > de.clientWidth + 1 && r.width > 0) {
            wide.push(`${e.tagName.toLowerCase()}.${String(e.className).slice(0, 40)}`);
            if (wide.length >= 5) break;
          }
        }
      }
      const text = (document.body.innerText + " " + document.documentElement.outerHTML).toLowerCase();
      return {
        overflowPx: de.scrollWidth - de.clientWidth,
        wide,
        broken,
        pending,
        images: document.images.length,
        sections: [...document.querySelectorAll("[data-section]")].map((e) => e.getAttribute("data-section")),
        h1: document.querySelectorAll("h1").length,
        title: document.title,
        termHits: terms.filter((t) => text.includes(t.toLowerCase())),
        hrefs: [...document.querySelectorAll("a[href]")].map((a) => a.getAttribute("href")),
      };
    }, TERMS);
    for (const h of data.hrefs) hrefs.add(h);
    if (shotsDir) await page.screenshot({ path: `${shotsDir}/${width}${route.replace(/\//g, "_") || "_home"}.png` });
    rows.push({ width, route, status, loadWide, docHeight, overflowPx: data.overflowPx, wide: data.wide, broken: data.broken, pending: data.pending, images: data.images, sections: data.sections.length, h1: data.h1, title: data.title, consoleErrors, pageErrors, failed, offOrigin: [...offOrigin], termHits: data.termHits });
    await page.close();
  }
  await context.close();
}

// internal link crawl
const links = { ok: 0, bad: [], external: [], other: [] };
const ctx = await browser.newContext();
for (const h of [...hrefs].sort()) {
  if (h.startsWith("#")) continue;
  if (h.startsWith("mailto:") || h.startsWith("tel:")) {
    links.other.push(h);
    continue;
  }
  if (/^https?:\/\//.test(h) && !h.startsWith(origin)) {
    links.external.push(h);
    continue;
  }
  const r = await ctx.request.get(new URL(h, base).href);
  if (r.status() === 200) links.ok++;
  else links.bad.push(`${h} → ${r.status()}`);
}
await browser.close();

const expect404 = (r) => (r.route === "/no-such-page" ? r.status === 404 : r.status === 200);
const problems = rows.filter((r) => !expect404(r) || r.loadWide || r.overflowPx > 0 || r.broken.length || r.pending || r.consoleErrors.filter((e) => !(r.route === "/no-such-page" && /404/.test(e))).length || r.pageErrors.length || r.failed.length || r.offOrigin.filter((o) => !ALLOW_OFF.includes(o)).length || r.termHits.length || r.h1 !== 1);
writeFileSync(outFile, JSON.stringify({ base, widths: WIDTHS, routes: ROUTES, rows, links, problems }, null, 1));
console.log(`pages checked: ${rows.length} (${ROUTES.length} routes x ${WIDTHS.length} widths)`);
console.log(`links: ${links.ok} internal ok, ${links.bad.length} bad, ${links.external.length} external, other: ${[...new Set(links.other)].join(" ")}`);
for (const b of links.bad) console.log("  BAD LINK " + b);
for (const e of links.external) console.log("  EXTERNAL " + e);
console.log(`problem rows: ${problems.length}`);
for (const p of problems) console.log("  " + JSON.stringify({ w: p.width, r: p.route, s: p.status, loadWide: p.loadWide, ov: p.overflowPx, wide: p.wide, broken: p.broken.slice(0, 3), pending: p.pending, ce: p.consoleErrors.slice(0, 2), pe: p.pageErrors.slice(0, 2), failed: p.failed.slice(0, 2), off: p.offOrigin, terms: p.termHits, h1: p.h1 }));

// Scratch tooling (not a repo file): BoostChat functional QA for boost-interior-demo-02.
//
// The tenant's widget allow-list holds https://interior-demo.boostweb.co.kr only, so the widget tears itself
// down on http://127.0.0.1:4322. This script loads the Demo 02 package UNDER the allow-listed origin: every
// request to that origin is answered from the local preview server (nothing reaches the real host), while
// boostchat.co.kr is the real production service. No BoostChat configuration is written.
//
// SAFETY: every /api/widget/<key>/lead request is answered here (marked "x-qa-stub: 1"); a lead response without
// the mark aborts the run. Page loads are paced (<= 3 per 65 s) to stay inside the widget's bootstrap budget.
//
//   node widget-qa.mjs <outDir> [phase...]      phases: local desktop contact mobile tablet
import { chromium, webkit, devices } from "/Users/woops/projects/web-recon-track-b/node_modules/playwright/index.mjs";
import { mkdirSync, writeFileSync } from "node:fs";

const ORIGIN = "https://interior-demo.boostweb.co.kr";
const CHAT = "https://boostchat.co.kr";
const KEY = "wgt_99kYYFOm7ABvdQbVh_8SdrnOlLrPqDI3";

const phases = phaseArgs.length ? phaseArgs : ["local", "desktop", "contact", "mobile", "tablet"];

const result = { startedAt: new Date().toISOString(), checks: [], phases: {} };
const check = (name, ok, detail) => {
  result.checks.push({ name, ok: !!ok, detail });
  console.log(`${ok ? "PASS" : "FAIL"}  ${name}${detail === undefined ? "" : "  " + JSON.stringify(detail).slice(0, 260)}`);
};
const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
const loads = [];
const pace = async (label) => {
  for (;;) {
    const now = Date.now();
    const recent = loads.filter((t) => now - t < 65000);
    if (recent.length < 3) break;
    const wait = 65000 - (now - recent[0]) + 500;
    console.log(`      … pacing: ${(wait / 1000).toFixed(1)} s before ${label}`);
    await sleep(wait);
  }
  loads.push(Date.now());
};

/** context wiring: local bytes under the allow-listed origin, lead stub, request log */
const wire = async (ctx, log, leadReply) => {
  await ctx.route(`${ORIGIN}/**`, async (route) => {
    const u = new URL(route.request().url());
    const response = await route.fetch({ url: LOCAL + u.pathname + u.search });
    await route.fulfill({ response });
  });
  // registered last = consulted first
  await ctx.route((url) => /\/api\/widget\/[^/]+\/lead/.test(url.pathname), async (route) => {
    const req = route.request();
    const cors = { "access-control-allow-origin": req.headers()["origin"] ?? ORIGIN, vary: "Origin", "cache-control": "no-store", "x-qa-stub": "1" };
    if (req.method() === "OPTIONS") return route.fulfill({ status: 204, headers: { ...cors, "access-control-allow-methods": "POST", "access-control-allow-headers": "content-type" } });
    const reply = leadReply(req);
    log.leads.push({ method: req.method(), url: req.url(), contentType: req.headers()["content-type"], body: req.postData(), replied: reply.status });
    return route.fulfill({ status: reply.status, headers: cors, contentType: "application/json", body: reply.body });
  });
};
const watch = (page, log) => {
  page.on("console", (m) => m.type() === "error" && log.consoleErrors.push(m.text().slice(0, 300)));
  page.on("pageerror", (e) => log.pageErrors.push(String(e).slice(0, 300)));
  page.on("requestfailed", (r) => log.failed.push(`${r.method()} ${r.url().slice(0, 140)} ${r.failure()?.errorText ?? ""}`));
  page.on("response", async (r) => {
    const u = r.url();
    if (r.status() >= 400) log.httpErrors.push(`${r.status()} ${r.request().method()} ${u.slice(0, 140)}`);
    if (/\/api\/widget\/[^/]+\/lead/.test(u) && r.headers()["x-qa-stub"] !== "1") {
      console.error("FATAL: an un-stubbed lead response was seen: " + u);
      process.exit(97);
    }
    if (u.startsWith(CHAT)) {
      const p = new URL(u).pathname;
      log.chat.push({ at: Date.now(), method: r.request().method(), path: p, status: r.status() });
      if (!p.includes(KEY) && p !== "/widget.js" && !p.startsWith("/_next/") && !/\.(js|css|woff2?|png|svg|ico|webp|jpg)$/.test(p)) log.otherKeyPaths.push(p);
    } else if (!u.startsWith(ORIGIN) && !u.startsWith(LOCAL) && !u.startsWith("data:") && !u.startsWith("blob:")) log.otherHosts.add(new URL(u).origin);
  });
};
const newLog = () => ({ consoleErrors: [], pageErrors: [], failed: [], httpErrors: [], chat: [], leads: [], otherKeyPaths: [], otherHosts: new Set() });
const sealLog = (log) => ({ ...log, otherHosts: [...log.otherHosts], chat: undefined, chatSummary: summarise(log.chat) });
const summarise = (rows) => {
  const m = new Map();
  for (const r of rows) {
    const k = `${r.method} ${r.path.replace(/\/_next\/.*/, "/_next/*")} ${r.status}`;
    m.set(k, (m.get(k) ?? 0) + 1);
  }
  return [...m].map(([k, n]) => `${n}x ${k}`);
};
const FRAME = `iframe[src*="/widget/${KEY}"]`;
const rect = async (loc) => {
  const b = await loc.boundingBox();
  return b ? { x: Math.round(b.x), y: Math.round(b.y), w: Math.round(b.width), h: Math.round(b.height) } : null;
};
const seat = (r, vp) => (r ? { right: vp.width - (r.x + r.w), bottom: vp.height - (r.y + r.h) } : null);
const overlap = (a, b) => {
  if (!a || !b) return null;
  const w = Math.min(a.x + a.w, b.x + b.w) - Math.max(a.x, b.x);
  const h = Math.min(a.y + a.h, b.y + b.h) - Math.max(a.y, b.y);
  return w > 0 && h > 0 ? { w, h, pctOfB: Math.round((100 * w * h) / (b.w * b.h)) } : { w: 0, h: 0, pctOfB: 0 };
};
/** wait until the newest assistant message stops changing */
const settledReply = async (frame, countBefore, maxMs = 90000) => {
  const msgs = frame.locator('[data-testid="chat-message"][data-role="assistant"]');
  const t0 = Date.now();
  let last = "";
  let stableSince = 0;
  while (Date.now() - t0 < maxMs) {
    const n = await msgs.count();
    if (n > countBefore) {
      const text = ((await msgs.nth(n - 1).innerText()) ?? "").trim();
      const body = text.split("\n").slice(1).join("\n").trim();
      if (body && !/^[.…\s]+$/.test(body) && text === last) {
        if (Date.now() - stableSince > 2500) return { text, ms: Date.now() - t0, count: n };
      } else {
        last = text;
        stableSince = Date.now();
      }
    }
    await sleep(400);
  }
  return { text: last, ms: Date.now() - t0, timedOut: true };
};

  page.on("request", (r) => isChat(r) && events.push([Date.now() - t0, "request"]));
  page.on("response", (r) => isChat(r.request()) && events.push([Date.now() - t0, "response " + r.status()]));
  page.on("requestfinished", (r) => isChat(r) && events.push([Date.now() - t0, "requestfinished"]));
  page.on("requestfailed", (r) => isChat(r) && events.push([Date.now() - t0, "requestfailed " + (r.failure()?.errorText ?? "")]));

  await page.goto(ORIGIN + "/portfolio", { waitUntil: "load" });

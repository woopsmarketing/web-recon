// Scratch tooling: list every Latin-letter run in rendered text and in a11y/placeholder attributes,
// so English neutral defaults that leaked into a Korean site are visible. Widget host is blocked.
//   node textscan.mjs <baseUrl> <route...>
import { chromium } from "/Users/woops/projects/web-recon-track-b/node_modules/playwright/index.mjs";
const [base, ...routes] = process.argv.slice(2);
const browser = await chromium.launch();
const seen = new Map();
for (const width of [1440, 390]) {
  const ctx = await browser.newContext({ viewport: { width, height: 900 }, isMobile: width < 700, hasTouch: width < 700, locale: "ko-KR" });
  await ctx.route("https://boostchat.co.kr/**", (r) => r.abort());
  for (const route of routes) {
    const page = await ctx.newPage();
    await page.goto(base + route, { waitUntil: "load" });
    await page.waitForTimeout(300);
    const found = await page.evaluate(() => {
      const out = [];
      const w = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT);
      while (w.nextNode()) {
        const p = w.currentNode.parentElement;
        if (!p || ["SCRIPT", "STYLE", "NOSCRIPT"].includes(p.tagName)) continue;
        const t = w.currentNode.nodeValue.trim();
        if (/[A-Za-z]{3,}/.test(t)) out.push(`text: ${t.slice(0, 90)}`);
      }
      for (const el of document.querySelectorAll("*")) {
        for (const a of ["aria-label", "alt", "title", "placeholder", "aria-roledescription", "value"]) {
          const v = el.getAttribute(a);
          if (v && /[A-Za-z]{3,}/.test(v)) out.push(`${a}: ${v.slice(0, 90)}`);
        }
      }
      out.push(`title: ${document.title}`);
      const d = document.querySelector('meta[name="description"]')?.getAttribute("content");
      if (d) out.push(`meta-description: ${d.slice(0, 120)}`);
      return out;
    });
    for (const f of found) seen.set(f, (seen.get(f) ?? new Set()).add(`${route}@${width}`));
    await page.close();
  }
  await ctx.close();
}
for (const [k, v] of [...seen].sort()) console.log(`${String(v.size).padStart(3)}  ${k}   [${[...v].slice(0, 2).join(", ")}]`);
await browser.close();

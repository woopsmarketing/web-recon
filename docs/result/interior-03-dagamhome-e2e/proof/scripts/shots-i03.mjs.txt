// Scratch tooling: full-page JPEG proof shots. node shots-i03.mjs <proofDir>
import { chromium } from "/Users/woops/projects/web-recon-track-b/node_modules/playwright/index.mjs";
import { mkdirSync } from "node:fs";
const proof = process.argv[2];
const SITES = [
  { dir: `${proof}/fixture`, base: "http://127.0.0.1:4340", details: ["sejong-dodam-34py-apartment", "daejeon-songchon-40py-townhouse"] },
  { dir: `${proof}/demo03-local`, base: "http://127.0.0.1:4341", details: ["suseong-white-34py-apartment-remodeling", "buk-32py-kitchen-bathroom-renewal"], block: true },
];
const browser = await chromium.launch();
let n = 0;
for (const s of SITES) {
  mkdirSync(s.dir, { recursive: true });
  const routes = ["/", "/portfolio", "/portfolio/page/2", ...s.details.map((d) => `/portfolio/${d}`), "/about", "/service", "/faq", "/contact", "/no-such-page"];
  const name = (r) => (r === "/" ? "home" : r.slice(1).replace(/\//g, "_"));
  for (const [w, mobile] of [[1440, false], [390, true], [768, false], [1024, false]]) {
    const list = w >= 768 && w < 1440 ? ["/", "/portfolio", "/contact"] : routes;
    const ctx = await browser.newContext({ viewport: { width: w, height: mobile ? 844 : 900 }, isMobile: mobile, hasTouch: mobile, locale: "ko-KR", timezoneId: "Asia/Seoul" });
    if (s.block) await ctx.route("**://boostchat.co.kr/**", (r) => r.abort("aborted"));
    for (const r of list) {
      if (s.block && r === "/portfolio/page/2" ) { /* 404 here; still shot */ }
      const p = await ctx.newPage();
      await p.goto(s.base + r, { waitUntil: "networkidle" });
      const h = await p.evaluate(() => document.documentElement.scrollHeight);
      for (let y = 0; y < h; y += 600) { await p.evaluate((y) => scrollTo(0, y), y); await p.waitForTimeout(70); }
      await p.waitForLoadState("networkidle"); await p.waitForTimeout(300);
      await p.evaluate(() => scrollTo(0, 0)); await p.waitForTimeout(300);
      await p.screenshot({ path: `${s.dir}/${name(r)}-${w}.jpg`, type: "jpeg", quality: 72, fullPage: true });
      n++; await p.close();
    }
    await ctx.close();
  }
}
await browser.close();
console.log("shots", n);

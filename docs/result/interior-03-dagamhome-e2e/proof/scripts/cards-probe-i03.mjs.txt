// Live probe: every portfolio recommendation card of the widget must lead to the site's own origin.
// Sends ONE chat message. For each card: open the viewer, read the project URL the viewer was given
// (network JSON from the widget API), go back to the chat. Finally click the LAST card's "open page"
// control with main-frame navigation recorded and aborted.
//   node cards-probe-i03.mjs --origin https://<host> --key wgt_… [--old-origin https://…]* [--out cards-probe.json]
//   (legacy positional form still works: node cards-probe-i03.mjs <origin> <oldOrigin> <out.json>)
//   --old-origin may be repeated or comma-separated. --key is only recorded (the page itself carries the key) and checked
//   against the widget iframe src. Talks to the REAL chat host and sends ONE chat message: do not run it before publish.
import { createRequire } from "node:module";
import { writeFileSync } from "node:fs";
const require = createRequire("/Users/woops/projects/web-recon-track-b/package.json");
const { chromium } = require("playwright");
const argv = process.argv.slice(2);
const named = {}; const pos = [];
for (let i = 0; i < argv.length; i++) { const m = /^--([^=]+)(?:=([\s\S]*))?$/.exec(argv[i]); if (m) { const v = m[2] ?? argv[++i]; (named[m[1]] ??= []).push(v); } else pos.push(argv[i]); }
const ORIGIN = (named.origin?.[0] ?? pos[0] ?? "").replace(/\/+$/, "");
const KEY = named.key?.[0] ?? null;
const OLDS = (named["old-origin"] ?? (pos[1] ? [pos[1]] : [])).flatMap((x) => x.split(",")).map((x) => x.trim().replace(/\/+$/, "")).filter(Boolean);
const OUT = named.out?.[0] ?? pos[2] ?? "cards-probe.json";
const SHOTS = named.shots?.[0] ?? null;
if (!/^https?:\/\/[^/]+$/.test(ORIGIN)) { console.error("cards-probe-i03: --origin <scheme://host> is required"); process.exit(64); }
const browser = await chromium.launch();
const ctx = await browser.newContext({ viewport: { width: 1440, height: 900 }, locale: "ko-KR", timezoneId: "Asia/Seoul" });
const page = await ctx.newPage();
const navs = []; let armed = false; const apiUrls = []; const result = { origin: ORIGIN, key: KEY, oldOrigin: OLDS.join(",") || null, oldOrigins: OLDS, chatMessagesSent: 0, cards: [], notes: [] };
await page.route("**/*", async (route) => {
  const req = route.request();
  if (armed && req.isNavigationRequest() && req.frame() === page.mainFrame()) { navs.push(req.url()); return route.abort(); }
  return route.continue();
});
ctx.on("page", async (p) => { await p.waitForLoadState("commit", { timeout: 8000 }).catch(() => {}); navs.push("POPUP " + p.url()); await p.close().catch(() => {}); });
// every absolute URL containing /portfolio/ that the widget API hands to the widget
page.on("response", async (res) => {
  const u = res.url();
  if (!u.startsWith("https://boostchat.co.kr/api/")) return;
  let body = ""; try { body = await res.text(); } catch { return; }
  for (const m of body.matchAll(/https?:\\?\/\\?\/[^"\\\s]+?\\?\/portfolio\\?\/[^"\\\s]*/g)) apiUrls.push({ api: new URL(u).pathname.replace(/wgt_[^/]+/, "<key>"), status: res.status(), url: m[0].replace(/\\\//g, "/") });
});
const dump = async (frame) => (await frame.locator("[data-testid]").evaluateAll((els) => [...new Set(els.map((e) => e.getAttribute("data-testid")))]).catch(() => [])).join(",");
try {
  await page.goto(ORIGIN + "/", { waitUntil: "load" });
  const frame = page.frameLocator("iframe[data-boost-chat-frame]").first();
  if (KEY) result.keyInIframeSrc = ((await page.locator("iframe[data-boost-chat-frame]").first().getAttribute("src", { timeout: 30000 }).catch(() => "")) ?? "").includes(KEY);
  await frame.locator('button[aria-label="상담창 열기"]').click({ timeout: 30000 });
  await frame.locator('[data-testid="chat-input"]').fill("화이트 톤 30평대 아파트 시공 사례를 보여주세요");
  result.sentAt = new Date().toISOString();
  await frame.locator('[data-testid="chat-send"]').click(); result.chatMessagesSent = 1;
  await frame.locator("[data-portfolio-card]").first().waitFor({ timeout: 60000 });
  await page.waitForTimeout(6000);
  if (SHOTS) { await frame.locator("[data-portfolio-card]").first().scrollIntoViewIfNeeded().catch(() => {}); await page.waitForTimeout(500); await page.screenshot({ path: SHOTS + "/chat-portfolio-cards-1440.jpg", type: "jpeg", quality: 82 }); }
  const more = frame.getByRole("button", { name: /더 보기/ });
  if (await more.count()) { await more.first().click().catch(() => {}); await page.waitForTimeout(800); }
  const n = await frame.locator("[data-portfolio-card]").count(); result.cardCount = n;
  for (let i = 0; i < n; i++) {
    const card = frame.locator("[data-portfolio-card]").nth(i);
    const entry = { index: i };
    try {
      entry.id = await card.getAttribute("data-portfolio-card", { timeout: 8000 });
      entry.title = ((await card.innerText()) || "").split("\n").slice(0, 3).join(" / ");
      const before = apiUrls.length;
      await card.locator('[data-testid="portfolio-card"]').click({ timeout: 8000 });
      await frame.locator('[data-testid="portfolio-viewer"]').waitFor({ timeout: 15000 });
      await frame.locator('[data-testid="portfolio-viewer-page"]').first().waitFor({ timeout: 15000 });
      if (SHOTS && i === 0) { await page.waitForTimeout(1200); await page.screenshot({ path: SHOTS + "/chat-portfolio-viewer-1440.jpg", type: "jpeg", quality: 82 }); }
      entry.viewerApiUrls = [...new Set(apiUrls.slice(before).map((x) => x.url))];
      if (i === n - 1) { // last card: really press "open page", record + abort the navigation
        armed = true; const nb = navs.length;
        await frame.locator('[data-testid="portfolio-viewer-page"]').first().click({ timeout: 8000 });
        for (let t = 0; t < 40 && navs.length === nb; t++) await page.waitForTimeout(250);
        entry.navigatedTo = navs.slice(nb);
      } else {
        await frame.locator('[data-testid="portfolio-viewer-back"]').first().click({ timeout: 8000 });
        await frame.locator('[data-testid="portfolio-viewer"]').waitFor({ state: "detached", timeout: 8000 }).catch(() => {});
        await page.waitForTimeout(500);
      }
    } catch (e) { entry.err = String(e).split("\n")[0].slice(0, 200); entry.testids = await dump(frame); }
    result.cards.push(entry);
  }
} catch (e) { result.fatal = String(e).split("\n")[0].slice(0, 300); }
const all = [...new Set(apiUrls.map((x) => x.url))];
result.apiPortfolioUrls = all;
result.apiPortfolioUrlSources = [...new Set(apiUrls.map((x) => x.api + " " + x.status))];
result.allApiUrlsOnOrigin = all.length > 0 && all.every((u) => u.startsWith(ORIGIN + "/"));
result.anyOld = all.some((u) => OLDS.some((o) => u.includes(o))) || navs.some((u) => OLDS.some((o) => u.includes(o)));
result.navs = navs;
writeFileSync(OUT, JSON.stringify(result, null, 1));
console.log(JSON.stringify(result, null, 1));
await browser.close();

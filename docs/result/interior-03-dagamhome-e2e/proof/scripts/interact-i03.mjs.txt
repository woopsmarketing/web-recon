    // ---------- 8. contact ----------
    {
      const raw = await (await ctx.request.get(site.base + "/contact")).text();

"").length, ...done });

    // ---------- 9. to-top + ext-widget ----------
    {

      if (vp.w === 1440) rec(site.extWidget ? "every page <html> has data-ext-widget" : "no page <html> has data-ext-widget", ext.every((e) => e.ext === site.extWidget), ext);
    }
    await ctx.close();
  }

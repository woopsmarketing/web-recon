/**
 * STEP 1 — live equals package (READ-ONLY, GET only).
 * For every file of the local package's site/ directory, GET the live URL with `Accept: * / *`
 * and compare sha256. X.html is served at /X, index.html at /. 404.html / _not-found.html are not
 * addressable: an unknown path must answer 404 with a body byte-identical to 404.html.
 *
 *   tsx --tsconfig platform/tsconfig.json live-parity.mts > proof/20-live-http-parity.log
 */
import { readFileSync, readdirSync, statSync } from "node:fs";
import { createHash } from "node:crypto";
import https from "node:https";
import path from "node:path";

const REPO = "/Users/woops/projects/web-recon-track-b";
const HOST = "interior-demo.boostweb.co.kr";
const EXPECTED_PACKAGE_HASH = "eeb82881a2fcf5f10dafa1148c1e4d3eb6aef84dd4afbfa8a2af3f8c6870e44d";
const pkgDir = path.join(REPO, "data/site-builds/boost-interior-demo/packages/384008316a39e1d58a0311af2410bcb20e57afabf1c2af6836fbfeb2bdbb0924");
const siteDir = path.join(pkgDir, "site");
const rec = JSON.parse(readFileSync(path.join(pkgDir, "build-record.json"), "utf8"));

function walk(d: string, rel = ""): string[] {
  let out: string[] = [];
  for (const e of readdirSync(d).sort()) {
    const p = path.join(d, e);
    const r = rel ? `${rel}/${e}` : e;
    if (statSync(p).isDirectory()) out = out.concat(walk(p, r));
    else out.push(r);
  }
  return out;
}
const files = walk(siteDir);

interface Res { status: number; headers: Record<string, string | string[] | undefined>; body: Buffer }
function get(p: string): Promise<Res> {
  return new Promise((resolve, reject) => {
    const r = https.request({ hostname: HOST, port: 443, method: "GET", path: p, headers: { host: HOST, accept: "*/*", "user-agent": "recon-readonly-verify" } }, (res) => {
      const chunks: Buffer[] = [];
      res.on("data", (c) => chunks.push(c));
      res.on("end", () => resolve({ status: res.statusCode ?? 0, headers: res.headers, body: Buffer.concat(chunks) }));
    });
    r.on("error", reject);
    r.end();
  });
}
const sha = (b: Buffer) => createHash("sha256").update(b).digest("hex");
const enc = (rel: string) => "/" + rel.split("/").map((s) => encodeURIComponent(s).replace(/%24/g, "$")).join("/");
function urlFor(rel: string): string | null {
  if (rel === "index.html") return "/";
  if (rel === "404.html" || rel === "_not-found.html") return null;
  if (rel.endsWith(".html")) return enc(rel.slice(0, -5));
  return enc(rel);
}

let ok = 0;
const mismatch: unknown[] = [];
const notAddressable: string[] = [];
const rows: string[] = [];
const byExt: Record<string, number> = {};
const queue = [...files];
async function worker() {
  while (queue.length) {
    const rel = queue.shift()!;
    const url = urlFor(rel);
    if (!url) { notAddressable.push(rel); continue; }
    const local = readFileSync(path.join(siteDir, rel));
    let res: Res | undefined;
    for (let a = 0; a < 3; a++) { try { res = await get(url); break; } catch (e) { if (a === 2) throw e; } }
    const lh = sha(local);
    const rh = sha(res!.body);
    const same = res!.status === 200 && lh === rh;
    if (same) { ok++; byExt[path.extname(rel) || "(none)"] = (byExt[path.extname(rel) || "(none)"] ?? 0) + 1; }
    else mismatch.push({ rel, url, status: res!.status, localLen: local.length, liveLen: res!.body.length, localSha: lh, liveSha: rh });
    rows.push(`${same ? "SAME" : "DIFF"}\t${res!.status}\t${local.length}\t${lh}\t${url}`);
  }
}
await Promise.all([worker(), worker(), worker(), worker()]);

console.log(`# STEP 1 live-vs-package parity — ${new Date().toISOString()}`);
console.log(`host: https://${HOST}   request header: Accept: */*   method: GET only`);
console.log(`package dir: ${path.relative(REPO, pkgDir)}`);
console.log(`build-record packageHash: ${rec.packageHash}  (expected ${EXPECTED_PACKAGE_HASH}: ${rec.packageHash === EXPECTED_PACKAGE_HASH ? "MATCH" : "DIFFERENT"})`);
console.log(`template: ${rec.template?.releaseId}`);
console.log(`files in site/: ${files.length}`);
console.log(`addressable files: ${files.length - notAddressable.length}`);
console.log(`byte-identical (status 200 + sha256 equal): ${ok} / ${files.length - notAddressable.length}`);
console.log(`identical by extension: ${JSON.stringify(byExt)}`);
console.log(`not addressable (checked through the 404 probe below): ${notAddressable.join(", ")}`);
console.log(`MISMATCH (${mismatch.length}): ${JSON.stringify(mismatch, null, 1)}`);

const local404 = readFileSync(path.join(siteDir, "404.html"));
const localNotFound = readFileSync(path.join(siteDir, "_not-found.html"));
console.log(`\nlocal 404.html sha256 ${sha(local404)} (${local404.length} B); local _not-found.html sha256 ${sha(localNotFound)} (${localNotFound.length} B); identical to each other: ${sha(local404) === sha(localNotFound)}`);
let probesOk = 0;
const probes = ["/definitely-not-a-page-qa-20261003", "/portfolio/definitely-not-a-record-qa-20261003", "/404.html", "/_not-found.html", "/404", "/_not-found"];
for (const p of probes) {
  const r = await get(p);
  const same = sha(r.body) === sha(local404);
  if (r.status === 404 && same) probesOk++;
  console.log(`404 probe ${p} → status ${r.status}, content-type ${r.headers["content-type"]}, cache-control ${r.headers["cache-control"]}, body sha256 ${sha(r.body)} (${r.body.length} B), body == 404.html: ${same}`);
}
console.log(`unknown-path probes answering 404 with the 404.html body: ${probesOk} / ${probes.length} (the first two are the required check; the rest record how the non-addressable names themselves answer)`);

console.log(`\nRESULT: ${mismatch.length === 0 && ok === files.length - notAddressable.length ? "PASS" : "FAIL"} — identical ${ok}/${files.length - notAddressable.length} addressable, ${notAddressable.length} non-addressable, mismatches ${mismatch.length}`);
console.log(`\n# per-file table (verdict, status, local bytes, local sha256, live path)`);
for (const r of rows.sort((a, b) => a.split("\t")[4]!.localeCompare(b.split("\t")[4]!))) console.log(r);

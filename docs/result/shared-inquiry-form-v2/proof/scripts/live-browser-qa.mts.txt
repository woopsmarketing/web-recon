/**
 * 31 — browser QA of the /contact inquiry form (interior-01 1.6.3) with the endpoint STUBBED.
 *
 *   ./node_modules/.bin/tsx --tsconfig platform/tsconfig.json docs/result/shared-inquiry-form-v2/proof/scripts/live-browser-qa.mts
 *   env: QA_BASE (unset = the live site | "local"), QA_PROFILES (comma list, default all six),
 *        QA_FAKE_LAUNCHER=1 (LOCAL only: a simulated launcher, to rehearse the widget check)
 *
 * NOTHING reaches https://boostchat.co.kr/api/widget/<key>/lead: every context gets the lead stub
 * (qa-lib installLeadStub — three routes, installed and self-tested before any page is opened) and
 * the process exits at once (code 97) if a lead response ever arrives without the stub's mark.
 * Live mode loads the real chat widget (its script, its iframe, its own bootstrap call) — that is
 * the point of the widget check; local mode blocks every third-party request.
 *
 * Per profile (Chromium 1440x900, 1024x768, 768x1024, 390x844, 375x667; WebKit iPhone 13), two page
 * loads of /contact:
 *   A  initial → validation (empty name; bad phone) → loading (answer held) → error + fallback
 *      contacts (network failure) → rate-limited (429, Retry-After 2: fallback contacts again,
 *      button disabled, then enabled again and the alert cleared) → conflict (409: no fallback
 *      contacts — the one failure without them) → success (200)
 *   B  keyboard (Tab order to the button) → 100-char name + 500-char message → Enter submits once
 * and, where the launcher is present, that its box never lies over the submit button, the alert, the
 * consent checkbox or the confirmation headline.
 * Result: proof/31-live-browser-qa.json (proof/rehearsal/… when local). Exit code 1 on FAIL.
 */

async function runProfile(key: string, profile: Profile) {
  const vw: number = profile.context.viewport.width;
  const phone = vw <= 390;
  const rec: Record<string, any> = { label: profile.label, engine: profile.engine, viewport: profile.context.viewport, states: {}, widget: { asserted: profile.widgetAsserted }, screenshots: [] as string[] };
  out.profiles[key] = rec;
  const browser = (browsers[profile.engine] ??= await pw[profile.engine].launch(profile.engine === "chromium" ? { args: localChromiumArgs(target) } : {}));
  out.tooling.browsers[profile.engine] = browser.version();
  const ctx = await browser.newContext({ ...profile.context, locale: "ko-KR", serviceWorkers: "block" });
  // the answers the stub gives, in order; a request nobody scripted is answered 500 and reported
  const script: Reply[] = [];
  const unscripted: number[] = [];
  const stub = await installLeadStub(ctx, target, site, (seen) => {
    const next = script.shift();
    if (next) return next;
    unscripted.push(seen.n);
    return { kind: "json", status: 500, body: '{"error":"qa_stub_unscripted"}', label: "UNSCRIPTED → 500" };
  });
  const page = await ctx.newPage();
  const log = attachLog(page);
  let navigations = 0;
  page.on("framenavigated", (f) => {
    if (f === page.mainFrame()) navigations++;
  });

  const st = () => state(page, site);
  const posts = () => stub.posts();
  const idOf = (n: number) => rawId(posts()[n]?.raw ?? null);
  const snap = async (name: string, fullPage = false) => {
    const file = `qa-${key}-${name}.jpg`;
    await page.screenshot({ path: path.join(SHOTS, file), type: "jpeg", quality: 70, fullPage });
    rec.screenshots.push(file);
  };
  const press = async (): Promise<string | null> => {
    try {
      await page.click("[data-inquiry-submit]", { timeout: 8000 });
      return null;
    } catch (e) {
      return String((e as Error).message).split("\n").slice(0, 3).join(" | ");
    }
  };
  const until = async (expr: string, timeout = 8000): Promise<boolean> => {
    try {
      await page.waitForFunction(expr, undefined, { timeout });
      return true;
    } catch {
      return false;
    }
  };
  const ALERT_SHOWN = `(document.querySelector("[data-inquiry-error]")?.textContent ?? "").length > 0`;
  const geometry = (s: any): Record<string, boolean> => {
    const a: Record<string, boolean> = {
      "no horizontal overflow": s.overflow.overflowX === false,
      "form inside the viewport width": within(s.formRect, s.viewport.w),
    };
    if (s.buttonRect) a["submit button inside the viewport width"] = within(s.buttonRect, s.viewport.w);
    if (s.alertVisible) {
      a["alert inside the viewport width"] = within(s.alertRect, s.viewport.w);
      a["alert text not clipped"] = s.alertClippedBy <= 1;
    }
    if (s.statusVisible) a["status region inside the viewport width"] = within(s.statusRect, s.viewport.w);
    if (phone && s.buttonRect) a["submit button at least 44 px tall (phone)"] = s.buttonRect.height >= 43.5;
    if (phone && s.contacts.links > 0) a["fallback links at least 44 px tall (phone)"] = s.contacts.heights.every((h: number) => h >= 43.5);
    return a;
  };
  const open = async (label: string) => {
    await gate(target, `${key} ${label}`);
    log.phase = label;
    const resp = await page.goto(`${target.base}/contact`, { waitUntil: "load", timeout: 45_000 });
    const mounted = await until(`(() => { const b = document.querySelector("[data-inquiry-submit]"); return !!b && b.disabled === false; })()`, 15_000);
    await page.evaluate(`document.fonts.ready.then(() => true)`).catch(() => {});
    if (FAKE) await callInPage(page, FAKE_LAUNCHER_JS);
    return { status: resp?.status() ?? null, mounted };
  };
  const keep = (name: string, s: any) => {
    rec.states[name] = s;
    return s;
  };

  // ---------------------------------------------------------------- widget
  const widgetExpected = target.mode === "live" || FAKE;
  let launcherSeen = false;
  const waitLauncher = async () => {
    const deadline = Date.now() + 10_000;
    for (;;) {
      const s = await st();
      if (s.launcher) return s.launcher;
      if (Date.now() >= deadline) return null;
      await sleep(300);
    }
  };
  /** the launcher against the form, at the position the page is in and again with `centre` in the middle of the viewport */
  const widgetAt = async (centre: string, pick: (s: any) => Record<string, boolean>) => {
    const natural = await st();
    await callInPage(page, CENTER_JS, centre);
    await sleep(200);
    const centred = await st();
    const assertions: Record<string, boolean> = {};
    for (const [where, s] of [["as the page left it", natural], [`${centre} centred`, centred]] as const) {
      if (!s.launcher) {
        assertions[`[${where}] launcher present`] = false;
        continue;
      }
      for (const [name, ok] of Object.entries(pick(s))) assertions[`[${where}] ${name}`] = ok;
    }
    const brief = (s: any) => ({ scrollY: s.scrollY, launcher: s.launcher?.rect ?? null, detectedBy: s.launcher?.detectedBy ?? null, buttonRect: s.buttonRect, alertRect: s.alertRect, consentRect: s.consentRect, doneTitleRect: s.doneTitleRect, buttonHit: s.buttonHit, launcherOver: s.launcherOver, launcherXRangeOverlapsButton: s.launcherXRangeOverlapsButton });
    return { assertions, detail: { natural: brief(natural), centred: brief(centred) } };
  };
  const widgetRow = (id: string, w: { assertions: Record<string, boolean>; detail: unknown }) => {
    rec.widget[id] = w.detail;
    if (profile.widgetAsserted) checks.row(key, id, w.assertions, w.detail);
    else say(`INFO  ${key}  ${id} — recorded only at this width (${Object.values(w.assertions).every(Boolean) ? "would pass" : "WOULD FAIL: " + Object.entries(w.assertions).filter(([, ok]) => !ok).map(([n]) => n).join("; ")})`);
  };

  try {
    // ================================================================ load A — the states
    const a = await open("load-A");
    const initial = keep("initial", await st());
    checks.row(key, "load: /contact answers 200 and the form is ready", { "status 200": a.status === 200, "submit button enabled once the form is mounted": a.mounted && initial.button?.disabled === false });
    if (widgetExpected) {
      const launcher = await waitLauncher();
      launcherSeen = launcher !== null;
      rec.widget.launcher = launcher;
      if (!launcherSeen) checks.note(`${key}: the chat launcher did not appear within 10 s of the page load — widget checks not run at this profile`);
    } else {
      checks.skip(key, "widget: launcher vs submit button / alert / consent checkbox / confirmation", "local run — the widget script is blocked");
    }

    log.phase = "initial";
    await callInPage(page, CENTER_JS, "[data-inquiry-submit]");
    await sleep(150);
    const init2 = keep("initial (button in view)", await st());
    checks.row(key, "initial", {
      [`button says "${copy.submitLabel}"`]: initial.button?.text === copy.submitLabel,
      "no alert, no confirmation": initial.alertVisible === false && initial.statusVisible === false && initial.doneTitleText === null,
      "9 form controls (7 fields + consent + trap), all empty": initial.controls === 9 && Object.values(initial.values).every((v) => v === "") && initial.consent === false && initial.trap === "",
      "no lead request yet": posts().length === 0,
      "the submit button is the top-most element at its own centre": init2.buttonHit?.[0] === "self",
      ...geometry(init2),
    });
    await snap("initial");
    if (launcherSeen) {
      widgetRow("widget: launcher vs form (as loaded)", await widgetAt("[data-inquiry-submit]", (s) => ({
        "launcher does not intersect the submit button": s.launcherOver.button === null,
        "launcher does not intersect the consent checkbox": s.launcherOver.consentCheckbox === null,
        "elementFromPoint(centre of the submit button) is the button": s.buttonHit?.[0] === "self",
      })));
    }

    // ---- validation: nothing may reach the endpoint
    log.phase = "validation";
    const e1 = await press();
    await sleep(400);
    const v1 = keep("validation: empty name", await st());
    checks.row(key, "validation: empty name", {
      "the press went through": e1 === null,
      "name reported missing by the browser": v1.validity.name?.valueMissing === true,
      "focus moved to the name field": v1.focus === "name",
      "no lead request": posts().length === 0,
      "no alert": v1.alertVisible === false,
      ...geometry(v1),
    });
    await snap("validation-empty-name");

    for (const [k, val] of Object.entries(VALUES)) await page.fill(`[name=${k}]`, k === "phone" ? "123" : val);
    await page.check("[name=consent]");
    const e2 = await press();
    await sleep(400);
    const v2 = keep("validation: bad phone", await st());
    checks.row(key, "validation: bad phone", {
      "the press went through": e2 === null,
      "phone reported invalid by the browser": v2.validity.phone?.valid === false && v2.validity.name?.valid === true,
      "focus moved to the phone field": v2.focus === "phone",
      "no lead request": posts().length === 0,
      "what was typed is kept": v2.values.name === VALUES.name && v2.values.message === VALUES.message && v2.values.phone === "123" && v2.consent === true,
      ...geometry(v2),
    });
    await snap("validation-bad-phone");
    await page.fill("[name=phone]", VALUES.phone);

    // ---- loading: the stub holds the answer
    log.phase = "loading";
    script.push(REPLY.hold());
    const e3 = await press();
    for (let i = 0; i < 50 && stub.heldCount() < 1; i++) await sleep(100);
    await until(`document.querySelector("[data-inquiry-submit]")?.disabled === true`, 4000);
    const loading = keep("loading", await st());
    await snap("loading");
    await callInPage(page, PRESS_AGAIN_JS);
    await sleep(350);
    checks.row(key, "loading (answer held)", {
      "the press went through": e3 === null,
      "exactly one POST, held by the stub": posts().length === 1 && stub.heldCount() === 1,
      "button disabled": loading.button?.disabled === true,
      [`button says "${copy.submittingLabel}"`]: loading.button?.text === copy.submittingLabel,
      "form is aria-busy": loading.ariaBusy === "true",
      "a second press while loading sends nothing": posts().length === 1 && unscripted.length === 0,
      ...geometry(loading),
    });

    // ---- error + fallback contacts: the held request fails on the network
    log.phase = "error";
    const navBefore = navigations;
    await stub.release(REPLY.network());
    await until(ALERT_SHOWN);
    await sleep(250);
    const failed = keep("error + fallback contacts", await st());
    checks.row(key, "error + fallback contacts (network failure)", {
      [`alert says "${copy.failureText}"`]: failed.alertOwnText === copy.failureText,
      [`fallback contacts offered under "${copy.fallbackLead}"`]: failed.contacts.present === true && failed.contacts.leadText === copy.fallbackLead && failed.contacts.links >= 1,
      "alert fully inside the viewport, top-most at its own centre": failed.alertInViewport === true && failed.alertHit?.[0] === "self",
      "every field keeps what was typed, consent still ticked": failed.values.name === VALUES.name && failed.values.phone === VALUES.phone && failed.values.region === VALUES.region && failed.values.area === VALUES.area && failed.values.schedule === VALUES.schedule && failed.values.message === VALUES.message && failed.consent === true && failed.controls === 9,
      [`button enabled again, says "${copy.submitLabel}"`]: failed.button?.disabled === false && failed.button?.text === copy.submitLabel,
      "focus not moved to the alert": failed.focus !== "alert" && failed.focus !== "status",
      "no navigation, still /contact, no confirmation": navigations === navBefore && failed.path === "/contact" && failed.doneTitleText === null,
      "still exactly one POST (nothing retried by itself)": posts().length === 1,
      ...geometry(failed),
    });
    await snap("error-fallback");
    if (launcherSeen) {
      widgetRow("widget: launcher vs form (failure alert visible)", await widgetAt("[data-inquiry-submit]", (s) => ({
        "failure alert visible": s.alertVisible === true && (s.alertOwnText ?? "").length > 0,
        "launcher does not intersect the submit button": s.launcherOver.button === null,
        "launcher does not intersect the alert": s.launcherOver.alert === null,
        "launcher does not intersect the consent checkbox": s.launcherOver.consentCheckbox === null,
        "elementFromPoint(centre of the submit button) is the button": s.buttonHit?.[0] === "self",
      })));
      await snap("widget-failure-button-centred");
    }

    // ---- rate-limited: 429 rate_limited, Retry-After 2
    log.phase = "rate-limited";
    const rateText = copy.rateLimitedText!.split("{minutes}").join("1");
    script.push(REPLY.rateLimited(2));
    const e4 = await press();
    await until(`(document.querySelector("[data-inquiry-error]")?.textContent ?? "").includes(${JSON.stringify(rateText)})`);
    const pausedAt = Date.now();
    const limited = keep("rate-limited", await st());
    await snap("rate-limited");
    await callInPage(page, PRESS_AGAIN_JS);
    await sleep(250);
    const postsWhilePaused = posts().length;
    checks.row(key, "rate-limited (429, Retry-After 2)", {
      "the press went through": e4 === null,
      [`alert says "${rateText}"`]: limited.alertOwnText === rateText,
      [`fallback contacts offered under "${copy.fallbackLead}" during the pause`]: limited.contacts.present === true && limited.contacts.leadText === copy.fallbackLead && limited.contacts.links >= 1,
      "button disabled during the pause": limited.button?.disabled === true,
      "the retry carried the SAME submission_id as the failed request": posts().length >= 2 && idOf(1) !== undefined && idOf(1) === idOf(0),
      "a press during the pause sends nothing": postsWhilePaused === 2 && unscripted.length === 0,
      "fields kept": limited.values.name === VALUES.name && limited.values.message === VALUES.message && limited.consent === true,
      ...geometry(limited),
    });
    const reenabled = await until(`(() => { const b = document.querySelector("[data-inquiry-submit]"); return !!b && b.disabled === false && (document.querySelector("[data-inquiry-error]")?.textContent ?? "") === ""; })()`, 10_000);
    const afterPause = keep("rate-limit pause over", await st());
    rec.pauseMs = Date.now() - pausedAt;
    checks.row(key, "rate-limit pause over", {
      "button enabled again": reenabled && afterPause.button?.disabled === false,
      "alert cleared": afterPause.alertText === "" && afterPause.alertVisible === false,
      "nothing was sent by itself": posts().length === 2,
      "took about the 2 s the endpoint asked for (not the 60 s default)": rec.pauseMs < 9000,
      ...geometry(afterPause),
    });

    // ---- conflict: 409
    log.phase = "conflict";
    script.push(REPLY.conflict());
    const e5 = await press();
    await until(ALERT_SHOWN);
    await sleep(250);
    const conflict = keep("conflict", await st());
    const leaked = ["idempotency", "409", "conflict", "submission"].filter((w) => String(conflict.alertText ?? "").toLowerCase().includes(w));
    checks.row(key, "conflict (409)", {
      "the press went through": e5 === null,
      [`alert says "${copy.conflictText}"`]: conflict.alertOwnText === copy.conflictText,
      "no raw vocabulary in the alert, no fallback contacts": leaked.length === 0 && conflict.contacts.present === false,
      "button enabled, fields kept": conflict.button?.disabled === false && conflict.values.name === VALUES.name && conflict.values.message === VALUES.message,
      "exactly three POSTs so far, the third with the SAME submission_id": posts().length === 3 && idOf(2) !== undefined && idOf(2) === idOf(0),
      ...geometry(conflict),
    });
    await snap("conflict");

    // ---- success: 200 {"received":true}
    log.phase = "success";
    script.push(REPLY.ok());
    const e6 = await press();
    const done = await page.waitForSelector(".i1-form__status--done", { timeout: 8000 }).then(() => true, () => false);
    await sleep(500);
    const success = keep("success", await st());
    checks.row(key, "success (200)", {
      "the press went through": e6 === null,
      [`confirmation "${copy.successTitle}" + body shown`]: done && success.doneTitleText === copy.successTitle && String(success.statusText ?? "").includes(copy.successBody!),
      "form fields removed": success.controls === 0 && success.button === null,
      "focus on the confirmation": success.focus === "status",
      "confirmation headline inside the viewport, below the header, top-most at its centre": success.doneTitleInViewport === true && success.doneTitleHit?.[0] === "self",
      "still /contact, no navigation": success.path === "/contact" && navigations === navBefore,
      "exactly four POSTs in this page load; after the conflict a NEW submission_id": posts().length === 4 && idOf(3) !== undefined && idOf(3) !== idOf(0),
      "no request the stub was not told to expect": unscripted.length === 0,
      ...geometry(success),
    });
    await snap("success");
    if (launcherSeen) {
      widgetRow("widget: launcher vs confirmation (success visible)", await widgetAt(".i1-form__done-title", (s) => ({
        "confirmation headline visible": s.doneTitleText === copy.successTitle && s.doneTitleRect !== null,
        "launcher does not intersect the confirmation headline": s.launcherOver.doneTitle === null,
      })));
    }

    // ================================================================ load B — keyboard, long content
    const b = await open("load-B");
    const postsBeforeB = posts().length;
    log.phase = "keyboard";
    // WebKit follows the Safari convention: plain Tab stops at text fields and menus only, and
    // Option+Tab stops at every control — so there the full order is walked with Alt+Tab, and what
    // plain Tab does is recorded next to it.
    const walk = async (chord: string) => {
      await page.focus("[name=name]");
      const order: (string | null)[] = [];
      for (let i = 0; i < TAB_ORDER.length; i++) {
        await page.keyboard.press(chord);
        order.push(await callInPage<string | null>(page, FOCUS_JS));
      }
      return order;
    };
    const chord = profile.engine === "webkit" ? "Alt+Tab" : "Tab";
    const plainOrder = profile.engine === "webkit" ? await walk("Tab") : null;
    const order = await walk(chord);
    rec.tabOrder = { chord, order, plainTabOrder: plainOrder };
    checks.row(key, `keyboard: ${chord} order name → phone → region → area → work type → schedule → message → consent → submit`, {
      "page reloaded and ready": b.status === 200 && b.mounted,
      [`order is ${TAB_ORDER.join(" → ")}`]: JSON.stringify(order) === JSON.stringify(TAB_ORDER),
      ...(plainOrder ? { "plain Tab walks the text fields and the menu in the same order (phone … message)": JSON.stringify(plainOrder.slice(0, 6)) === JSON.stringify(TAB_ORDER.slice(0, 6)) } : {}),
